cmake_minimum_required(VERSION 3.24)
project(sk150_control VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SK150_FETCH_WX "Build wxWidgets from its official source" OFF)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_compile_definitions(NOMINMAX)
find_package(Threads REQUIRED)
if(SK150_FETCH_WX)
  include(FetchContent)
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(wxWidgets
    GIT_REPOSITORY https://github.com/wxWidgets/wxWidgets.git
    GIT_TAG v3.3.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(wxWidgets)
  set(SK150_WX_LIBS wx::core wx::base)
else()
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
  include(${wxWidgets_USE_FILE})
  set(SK150_WX_LIBS ${wxWidgets_LIBRARIES})
endif()
add_library(sk150_protocol src/modbus.cpp)
target_include_directories(sk150_protocol PUBLIC src)
target_link_libraries(sk150_protocol PUBLIC Threads::Threads)
add_executable(sk150 WIN32 src/main.cpp)
if(WIN32)
  enable_language(RC)
  target_sources(sk150 PRIVATE src/windows.rc)
endif()
target_link_libraries(sk150 PRIVATE sk150_protocol ${SK150_WX_LIBS})
if(MINGW)
  target_link_options(sk150 PRIVATE -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol-test tests/protocol_test.cpp)
  target_link_libraries(protocol-test PRIVATE sk150_protocol)
  add_test(NAME protocol COMMAND protocol-test)
  add_executable(connection-test tests/connection_test.cpp)
  target_include_directories(connection-test PRIVATE src)
  add_test(NAME connection COMMAND connection-test)
  add_executable(serial-path-test tests/serial_path_test.cpp)
  target_include_directories(serial-path-test PRIVATE src)
  add_test(NAME serial-path COMMAND serial-path-test)
  add_executable(display-test tests/display_test.cpp)
  target_link_libraries(display-test PRIVATE sk150_protocol ${SK150_WX_LIBS})
  add_test(NAME display COMMAND display-test)
endif()
